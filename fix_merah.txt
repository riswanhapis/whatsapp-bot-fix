def kirim_email(nomor):

    pesan = f"""
Halo Tim WhatsApp,

Perkenalkan, saya Epep. Saya ingin mengajukan banding atau permohonan peninjauan ulang terkait kendala yang saya alami saat mencoba mendaftarkan nomor telepon saya ke aplikasi WhatsApp.

Nomor telepon: +{nomor}

Atas perhatian dan bantuan dari pihak WhatsApp, saya ucapkan terima kasih.

Hormat saya,

"""

    msg = MIMEText(pesan)
    msg['Subject'] = "Request Review WhatsApp Account"
    msg['From'] = EMAIL
    msg['To'] = "support@support.whatsapp.com"

    server = smtplib.SMTP_SSL('smtp.gmail.com', 465)
    server.login(EMAIL, APP_PASSWORD)

    server.quit()

# ========================
# BOT COMMANDS
# ========================

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):

    await update.message.reply_text(
"""🔧 Fix Nomor WhatsApp

Kirim nomor yang mengalami masalah.

Format:
628xxxxxxxxxx
"""
)

async def kirim(update: Update, context: ContextTypes.DEFAULT_TYPE):

    nomor = update.message.text

    await update.message.reply_text("⏳ Mengirim email ke WhatsApp Support...")

    kirim_email(nomor)

    tanggal = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    cursor.execute(
        "INSERT INTO requests (nomor,tanggal) VALUES (?,?)",
        (nomor, tanggal)
    )

    conn.commit()

    await update.message.reply_text(
f"""✅ Email berhasil dikirim

Nomor: {nomor}

Silakan tunggu balasan dari WhatsApp Support."""
)

# ========================
# ADMIN PANEL
# ========================

async def panel(update: Update, context: ContextTypes.DEFAULT_TYPE):

    if update.effective_user.id != ADMIN_ID:
        return

    cursor.execute("SELECT COUNT(*) FROM requests")

    total = cursor.fetchone()[0]

    await update.message.reply_text(
f"""📊 Panel Admin

Total Request: {total}
"""
)

# ========================
# LIST DATA
# ========================

async def list_nomor(update: Update, context: ContextTypes.DEFAULT_TYPE):

    if update.effective_user.id != ADMIN_ID:
        return

    cursor.execute("SELECT nomor,tanggal FROM requests")

    rows = cursor.fetchall()

    if not rows:
        await update.message.reply_text("Database kosong")
        return

    text = "📂 Daftar Nomor\n\n"

    for nomor, tanggal in rows:
        text += f"{nomor} | {tanggal}\n"

    await update.message.reply_text(text)

# ========================
# START BOT
# ========================

app = ApplicationBuilder().token(TOKEN).build()

app.add_handler(CommandHandler("start", start))
app.add_handler(CommandHandler("panel", panel))
app.add_handler(CommandHandler("list", list_nomor))
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, kirim))

print("Bot running...")

app.run_polling()
